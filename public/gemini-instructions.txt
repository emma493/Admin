PROJECT OVERVIEW:
The Shortxx Admin Dashboard backend now operates two isolated tabs and collections:

1. "Videos" Tab (Main Feed):
   - Targets the "videos" collection in Firestore.
   - Holds direct video stream files (.mp4, .m3u8, .webm, CDN).
   - Feeds the main player via vid.js.
   - UNCHANGED & NOT TOUCHED.

2. "Ads" Tab (XVideos Iframe Embeds):
   - Targets the "ads" collection in Firestore.
   - Converts XVideos watch/embed links to canonical iframe embed URLs.
   - Feeds the dedicated ad section rendered in ads.html.

--------------------------------------------------------------------------------
FIRESTORE "ads" DOCUMENT STRUCTURE
--------------------------------------------------------------------------------
Each document stored in the "ads" collection has the following format:
{
  "id": "FIRESTORE_AUTO_DOC_ID",
  "is_active": true,
  "type": "iframe",
  "direct_url": "https://www.xvideos.com/embedframe/{VIDEO_ID}",
  "views": 0,
  "created_at": "SERVER_TIMESTAMP"
}

--------------------------------------------------------------------------------
FRONTEND INTEGRATION INSTRUCTIONS (FOR ads.html)
--------------------------------------------------------------------------------
1. Initialize Firestore with the named database:
   const app = initializeApp({ projectId: "intense-augury-28gvj" });
   const db = getFirestore(app, "ai-studio-shortxxadmindash-86192a98-919e-436c-80b9-836d96e0e32b");

2. Query active ads:
   const q = query(collection(db, "ads"), where("is_active", "==", true));
   const snapshot = await getDocs(q);
   const ads = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));

3. Render as an <iframe> (NOT a <video> element):
   const chosenAd = ads[Math.floor(Math.random() * ads.length)];
   document.getElementById("ad-iframe").src = chosenAd.direct_url;

4. Track views atomically in Firestore:
   await updateDoc(doc(db, "ads", chosenAd.id), {
     views: increment(1)
   });

READY-TO-USE ads.html CODE (COPY & PASTE INTO FRONTEND PROJECT)
--------------------------------------------------------------------------------

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1.0" />
  <title>Shortxx Ads</title>
  <style>
    * { margin: 0; padding: 0; box-sizing: border-box; }
    body, html { width: 100%; height: 100%; background: #000; overflow: hidden; }
    #wrapper { position: relative; width: 100%; height: 100%; display: flex; align-items: center; justify-content: center; }
    iframe { width: 100%; height: 100%; border: none; }
    #badge { position: absolute; top: 12px; left: 12px; z-index: 10; background: rgba(0,0,0,0.8); color: #ff3b30; font-size: 11px; font-weight: 900; padding: 4px 8px; border-radius: 6px; border: 1px solid rgba(255,59,48,0.4); }
    #next-btn { position: absolute; bottom: 20px; right: 20px; z-index: 10; background: #dc2626; color: #fff; border: none; padding: 10px 16px; border-radius: 9999px; font-size: 13px; font-weight: bold; cursor: pointer; }
  </style>
</head>
<body>
  <div id="wrapper">
    <div id="badge">SPONSORED AD</div>
    <iframe id="ad-iframe" src="about:blank" frameborder="0" allowfullscreen allow="autoplay; fullscreen"></iframe>
    <button id="next-btn" onclick="nextAd()">Next Ad →</button>
  </div>

  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js";
    import { getFirestore, collection, query, where, getDocs, doc, updateDoc, increment } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js";

    const app = initializeApp({ projectId: "intense-augury-28gvj" });
    const db = getFirestore(app, "ai-studio-shortxxadmindash-86192a98-919e-436c-80b9-836d96e0e32b");

    let ads = [];
    let index = -1;

    async function init() {
      try {
        const q = query(collection(db, "ads"), where("is_active", "==", true));
        const snap = await getDocs(q);
        ads = snap.docs.map(d => ({ id: d.id, ...d.data() }));
        if (ads.length > 0) nextAd();
      } catch(e) {
        console.error("Failed to load ads:", e);
      }
    }

    window.nextAd = async function() {
      if (!ads.length) return;
      index = (index + 1) % ads.length;
      const ad = ads[index];
      document.getElementById("ad-iframe").src = ad.direct_url;
      try {
        await updateDoc(doc(db, "ads", ad.id), { views: increment(1) });
      } catch(e) {}
    };

    window.addEventListener("DOMContentLoaded", init);
  </script>
</body>
</html>
